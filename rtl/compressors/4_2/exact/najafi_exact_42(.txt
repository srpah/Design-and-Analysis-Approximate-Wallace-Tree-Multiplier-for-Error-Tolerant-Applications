module najafi_exact_42(
    input  x1, x2, x3, x4,
    input  Cin,
    output Sum,
    output Carry,
    output Cout
);

    wire xor1, xor2, xor3;

    assign xor1 = x1 ^ x2;

    assign xor2 = xor1 ^ x3;

    assign xor3 = x4 ^ Cin;
    assign Cout = (x1 & x2) | (x2 & x3) | (x3 & x1);
    assign Sum = xor2 ^ xor3;
    assign Carry = (xor2 & x4) |
                   (x4 & Cin) |
                   (Cin & xor2);

endmodule
