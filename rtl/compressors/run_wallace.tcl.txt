# ============================================================
# run_wallace.tcl
# Complete Vivado flow for Wallace tree multiplier
# No FPGA hardware required — synthesis + implementation only
#
# Usage:
#   vivado -mode batch -source run_wallace.tcl
# or from Vivado Tcl console:
#   source run_wallace.tcl
# ============================================================

# ---- Configuration ----------------------------------------
set PROJECT_NAME  "wallace_8x8_project"
set PROJECT_DIR   "./vivado_output"
set SRC_FILE      "./wallace_8x8.v"
set XDC_FILE      "./wallace_8x8.xdc"
set PART          "xc7a35tcpg236-1"   ;# Artix-7 — change to your device
                                        ;# e.g. xc7z020clg400-1  (Zynq)
                                        ;#      xc7k325tffg900-2  (Kintex)
                                        ;#      xcvu9p-flgb2104-2-i (UltraScale+)
set TOP_MODULE    "wallace_8x8"
set CLK_PERIOD_NS 10.0                  ;# 100 MHz

# ---- Step 1: Create project --------------------------------
puts "\n=== [1/6] Creating project ==="
file mkdir $PROJECT_DIR
create_project $PROJECT_NAME $PROJECT_DIR -part $PART -force

# Add source files
add_files -norecurse $SRC_FILE
add_files -fileset constrs_1 -norecurse $XDC_FILE

# Set top module
set_property top $TOP_MODULE [current_fileset]
update_compile_order -fileset sources_1

# ---- Step 2: Synthesis -------------------------------------
puts "\n=== [2/6] Running synthesis ==="
synth_design \
    -top      $TOP_MODULE \
    -part     $PART \
    -flatten_hierarchy full \
    -keep_equivalent_registers \
    -no_lc

# Save synthesised checkpoint
write_checkpoint -force $PROJECT_DIR/post_synth.dcp
puts "Synthesis checkpoint saved."

# ---- Step 3: Synthesis reports (area + timing estimate) ----
puts "\n=== [3/6] Synthesis reports ==="
report_utilization \
    -file   $PROJECT_DIR/synth_utilization.rpt \
    -hierarchical
report_timing_summary \
    -file   $PROJECT_DIR/synth_timing.rpt \
    -max_paths 10 \
    -delay_type max
report_power \
    -file   $PROJECT_DIR/synth_power.rpt

puts "Synthesis reports written to $PROJECT_DIR/"

# ---- Step 4: Implementation (place and route) --------------
puts "\n=== [4/6] Running implementation ==="
opt_design
place_design
route_design

# Save routed checkpoint
write_checkpoint -force $PROJECT_DIR/post_route.dcp
puts "Implementation checkpoint saved."

# ---- Step 5: Post-route reports (most accurate) ------------
puts "\n=== [5/6] Post-route reports ==="

# TIMING REPORT — full detail
report_timing_summary \
    -file      $PROJECT_DIR/route_timing_summary.rpt \
    -max_paths 20 \
    -delay_type max \
    -report_unconstrained

report_timing \
    -file      $PROJECT_DIR/route_timing_detail.rpt \
    -max_paths 5 \
    -delay_type max \
    -sort_by    slack \
    -input_pins \
    -routable_nets

# AREA / UTILISATION REPORT
report_utilization \
    -file      $PROJECT_DIR/route_utilization.rpt \
    -hierarchical \
    -hierarchical_depth 3

# POWER REPORT — post-route is the most accurate estimate
report_power \
    -file      $PROJECT_DIR/route_power.rpt \
    -format    text \
    -verbose

# DESIGN ANALYSIS — logic levels and critical path
report_design_analysis \
    -file      $PROJECT_DIR/design_analysis.rpt \
    -max_paths 10 \
    -logic_level_distribution

# CLOCK INTERACTION
report_clock_interaction \
    -file      $PROJECT_DIR/clock_interaction.rpt

puts "Post-route reports written to $PROJECT_DIR/"

# ---- Step 6: Parse and print key numbers to console --------
puts "\n=== [6/6] Key results summary ==="

# Read timing summary and extract WNS / TNS
set timing_file [open $PROJECT_DIR/route_timing_summary.rpt r]
set timing_data [read $timing_file]
close $timing_file

# Extract Worst Negative Slack
if {[regexp {WNS\(ns\)\s+([-\d\.]+)} $timing_data match wns]} {
    puts "  Worst Negative Slack (WNS) : $wns ns"
    if {$wns >= 0} {
        set freq [expr {1000.0 / ($CLK_PERIOD_NS - $wns)}]
        puts "  Max achievable frequency   : [format %.1f $freq] MHz"
    } else {
        puts "  WARNING: timing NOT closed at ${CLK_PERIOD_NS} ns"
        set crit_path [expr {$CLK_PERIOD_NS - $wns}]
        puts "  Critical path delay        : [format %.3f $crit_path] ns"
    }
}

# Read utilisation and extract LUT / FF counts
set util_file [open $PROJECT_DIR/route_utilization.rpt r]
set util_data [read $util_file]
close $util_file

if {[regexp {Slice LUTs\s*\|\s*(\d+)} $util_data match luts]} {
    puts "  Slice LUTs used            : $luts"
}
if {[regexp {Slice Registers\s*\|\s*(\d+)} $util_data match ffs]} {
    puts "  Slice Registers (FFs)      : $ffs"
}
if {[regexp {DSP48E\d*\s*\|\s*(\d+)} $util_data match dsps]} {
    puts "  DSP blocks used            : $dsps  (should be 0)"
}
if {[regexp {Block RAM.*\|\s*(\d+)} $util_data match brams]} {
    puts "  Block RAMs used            : $brams"
}

# Read power report and extract total power
set pwr_file [open $PROJECT_DIR/route_power.rpt r]
set pwr_data [read $pwr_file]
close $pwr_file

if {[regexp {Total On-Chip Power \(W\)\s*\|\s*([\d\.]+)} $pwr_data match total_pwr]} {
    puts "  Total on-chip power        : $total_pwr W"
}
if {[regexp {Dynamic\s*\(W\)\s*\|\s*([\d\.]+)} $pwr_data match dyn_pwr]} {
    puts "  Dynamic power              : $dyn_pwr W"
}
if {[regexp {Static\s*\(W\)\s*\|\s*([\d\.]+)} $pwr_data match static_pwr]} {
    puts "  Static (leakage) power     : $static_pwr W"
}

puts "\n=== Flow complete. All reports in: $PROJECT_DIR/ ==="
puts "Reports written:"
puts "  synth_utilization.rpt       — post-synthesis area"
puts "  synth_timing.rpt            — post-synthesis timing estimate"
puts "  synth_power.rpt             — post-synthesis power estimate"
puts "  route_utilization.rpt       — post-route area (most accurate)"
puts "  route_timing_summary.rpt    — post-route timing summary"
puts "  route_timing_detail.rpt     — post-route critical path detail"
puts "  route_power.rpt             — post-route power (most accurate)"
puts "  design_analysis.rpt         — logic levels, path breakdown"
puts "  clock_interaction.rpt       — clock domain crossing check"

close_project
