module wgfa(input A,B,C, output carry,sum);
wire h1,h2;
assign h1=A|B;
assign h2= ~(A&B);
assign x=h1&h2;
assign sum = C?~x;x;
assign carry = (h1&x)|~h2;
endmodule

module sorter(input a,b, output max, min);
assign max = a|b;
assign min=a&b;
endmodule

module halfsort(input A,B,C,D, output A1,B1,C1,D1);
wire s1max, s1min, s2max,s2min,s3max,s3min,s4max,s4min;
sorter s1(.a(A),.b(B),.max(s1max),.min(s1min));
sorter s2(.a(C),.b(D),.max(s2max),.min(s2min));
sorter s3(.a(s1max),.b(s2max),.max(s3max),.min(s3min));
sorter s4(.a(s1min),.b(s2min),.max(s4max),.min(s4min));
A1=s3max;
B1=s4max;
C1=s3min;
D1=s4min;
endmodule
